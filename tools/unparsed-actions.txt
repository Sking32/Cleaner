            Save-OptimizationBackup 'gamethrottle' $path 'PowerThrottlingOff' 0 -Type 'Remove' -Category 'games'
            if ($Enable) {
                if (-not (Test-Path $path)) { New-Item -Path $path -Force | Out-Null }
                Set-ItemProperty -Path $path -Name 'PowerThrottlingOff' -Value 1 -Type DWord -Force
            } else {
                Remove-ItemProperty -Path $path -Name 'PowerThrottlingOff' -ErrorAction SilentlyContinue
            }
            return $true
        }
        GetState = {
            try {
                $v = Get-ItemProperty -Path 'HKLM:\SYSTEM\CurrentControlSet\Control\Power\PowerThrottling' -Name 'PowerThrottlingOff' -ErrorAction Stop
                return ([int]$v.PowerThrottlingOff -eq 1)
            } catch { return $false }
        }
